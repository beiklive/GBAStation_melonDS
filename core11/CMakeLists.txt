cmake_minimum_required(VERSION 3.16)
project(GBAStationCore11 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(GBAS "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(FULLDEX_SOURCE "" CACHE PATH "Checkout of pokemon-fulldex (c255f97b7be1e8b5ebe8f792cf01b69660d74016)")
if(NOT FULLDEX_SOURCE)
  include(FetchContent)
  FetchContent_Declare(fulldex GIT_REPOSITORY https://github.com/mamesj/pokemon-fulldex.git
    GIT_TAG c255f97b7be1e8b5ebe8f792cf01b69660d74016 GIT_SUBMODULES "")
  FetchContent_GetProperties(fulldex)
  if(NOT fulldex_POPULATED)
    FetchContent_Populate(fulldex)
  endif()
  set(FULLDEX_SOURCE "${fulldex_SOURCE_DIR}")
endif()
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT ON CACHE BOOL "" FORCE)
set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
add_subdirectory("${FULLDEX_SOURCE}/melonDS" melonDS EXCLUDE_FROM_ALL)
include(NativeRenderer.cmake)
add_library(session OBJECT Session.cpp "${FULLDEX_SOURCE}/frontend/platform.cpp")
target_link_libraries(session PRIVATE core)
target_include_directories(session PRIVATE "${FULLDEX_SOURCE}/frontend")
if(NINTENDO_SWITCH)
  include(Switch.cmake)
else()
  add_executable(replay Replay.cpp)
  target_link_libraries(replay PRIVATE session core)
  add_executable(render_thread_check EXCLUDE_FROM_ALL "${GBAS}/tests/core11_render_thread.cpp")
  target_include_directories(render_thread_check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(render_thread_check PRIVATE session core)
  add_executable(cheat_check EXCLUDE_FROM_ALL "${GBAS}/tests/core11_cheats.cpp"
    "${FULLDEX_SOURCE}/frontend/platform.cpp")
  add_library(cheat_database_check OBJECT EXCLUDE_FROM_ALL
    "${GBAS}/nds_stub/src/NdsCheatDatabase.cpp" "${GBAS}/src/CRC32.cpp")
  target_include_directories(cheat_database_check PRIVATE "${GBAS}/src" "${GBAS}/nds_stub/include")
  target_include_directories(cheat_check PRIVATE "${FULLDEX_SOURCE}/frontend" "${GBAS}/nds_stub/include")
  target_link_libraries(cheat_check PRIVATE core cheat_database_check)
endif()
set_source_files_properties("${FULLDEX_SOURCE}/frontend/platform.cpp" PROPERTIES
    COMPILE_DEFINITIONS "Mic_ReadInput=UnusedFrontendMicRead")
